// Exemple d'ús de WHILE

      let monedes = 0;
      const MONEDES_OBJECTIU = 100;
      const MONEDES_PER_ENEMIC = 15;
      let enemicsDerrotats = 0;

      // Derrotant enemics i Recollint monedes...;

      while (monedes < MONEDES_OBJECTIU) {
        enemicsDerrotats++;
        monedes += MONEDES_PER_ENEMIC;
        // Enemic Derrotat!
      }

      // Objectiu assolit!
      // Enemics derrotats: 7
      // Monedes totals: 105
